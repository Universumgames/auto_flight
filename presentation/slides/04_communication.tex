    \begin{column}{0.5\textwidth}
      \textbf{\faBroadcastTower\; Long-range telemetry}
      \begin{itemize}
        \item Plane $\leftrightarrow$ base over \hilite{LoRa} (RadioLib / SX1262)
        \item Eigenes Protokoll Layer (Transport und Application Layer)
        \item Überträgt Position, Sensoren, Routen \& Komponenten-Status
              Updates, ACKs and keep-alive pings
        \item \emph{Long range, low bandwidth} — nur essenzielle Daten
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{\faGlobe\; Web ground control}
      \begin{itemize}
        \item Base station öffnet einen \hilite{WiFi access point}
        \item Stellt ein Webinterface zur Verfügung \hilite{Vue 3 + TypeScript + Leaflet}
        \item Abzudeckender Bereich frei wählbar, Route kann direkt im Browser gezeichnet werden
        \item Live map, telemetry \& connection status über
              \hilite{WebSocket}
      \end{itemize}
