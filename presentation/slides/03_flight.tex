\begin{frame}{Autonomous flight}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \metroset{block=fill}
      \begin{block}{\faDrawPolygon\; Route planning}
        A drawn \hilite{polygon} is swept into a
        \hilite{lawn-mower route} by the \texttt{route\_planner} component.
      \end{block}
      \begin{block}{\faCompass\; Sensor fusion}
        GPS (position/alt), MPU6050 (attitude), magnetometer (heading)
        and barometer (altitude) feed the controller.
      \end{block}
      \begin{block}{\faSlidersH\; PID steering}
        Aileron, pitch and thrust are driven by \hilite{PID loops}
        toward the next waypoint.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
