\begin{frame}{The idea}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.58\textwidth}
      An \hilite{autonomous glider} that flies pre-planned coverage
      patterns over an area — a field, for example — so it can later carry a
      camera for \hilite{aerial surveying and crop analysis}.

      \bigskip
      \begin{itemize}
        \item Navigates itself from \alert{GPS}, an \alert{IMU}, a
              \alert{magnetometer} and a \alert{barometer}
        \item Stays in contact with a ground base station over a
              \alert{long-range LoRa} link
        \item Can be taken over \alert{manually via RC} in an emergency
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.95]
